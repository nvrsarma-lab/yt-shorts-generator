async function generateStory() { const topic = document.getElementById("topic").value.trim(); const language = document.getElementById("language").value; const sceneCount = Number( document.getElementById("sceneCount").value ); if (!topic) { alert("Please enter a story topic first."); return; } state.topic = topic; state.language = language; state.sceneCount = sceneCount; showStatus( "Creating your story...", true ); const button = document.getElementById( "generateStoryBtn" ); button.disabled = true; try { const response = await fetch( API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "story", topic: topic, language: language, sceneCount: sceneCount }) } ); const data = await response.json(); console.log( "STORY RESPONSE:", data ); if (!response.ok || !data.success) { throw new Error( data.error || "Story generation failed." ); } state.story = data.content || ""; document.getElementById( "storyOutput" ).textContent = state.story; document.getElementById( "storyCard" ).classList.add("show"); activateStep(2); showStatus( "Story generated successfully." ); document.getElementById( "storyCard" ).scrollIntoView({ behavior: "smooth" }); } catch (error) { console.error( "STORY ERROR:", error ); showStatus( error.message ); } finally { button.disabled = false; } }
