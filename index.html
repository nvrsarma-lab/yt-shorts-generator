async function generateStory() { // ============================================ // GET HTML ELEMENTS // ============================================ const topicElement = document.getElementById("topic"); const languageElement = document.getElementById("language"); const sceneCountElement = document.getElementById("sceneCount"); const button = document.getElementById("generateStoryBtn"); const storyOutput = document.getElementById("storyOutput"); const storyCard = document.getElementById("storyCard"); // ============================================ // CHECK REQUIRED ELEMENTS // ============================================ if (!topicElement) { alert("Topic input not found."); return; } if (!languageElement) { alert("Language selector not found."); return; } if (!sceneCountElement) { alert("Scene count selector not found."); return; } // ============================================ // READ USER INPUT // ============================================ const topic = topicElement.value.trim(); const language = languageElement.value; const sceneCount = Number( sceneCountElement.value ); // ============================================ // VALIDATE TOPIC // ============================================ if (!topic) { alert( "Please enter a story topic first." ); topicElement.focus(); return; } // ============================================ // VALIDATE SCENE COUNT // ============================================ if ( !Number.isInteger(sceneCount) || sceneCount < 1 ) { alert( "Please select a valid number of scenes." ); return; } // ============================================ // SAVE STATE // ============================================ if ( typeof state !== "undefined" ) { state.topic = topic; state.language = language; state.sceneCount = sceneCount; } // ============================================ // BUTTON LOADING // ============================================ if (button) { button.disabled = true; button.textContent = "⏳ Generating Story..."; } // ============================================ // STATUS // ============================================ if ( typeof showStatus === "function" ) { showStatus( "Creating your story...", true ); } // ============================================ // CLOUDFLARE WORKER URL // ============================================ const WORKER_URL = "https://yt-shorts-api.nvrsarma.workers.dev/"; try { // ========================================== // SEND REQUEST // ========================================== const response = await fetch( WORKER_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "story", topic: topic, language: language, sceneCount: sceneCount }) } ); // ========================================== // READ RESPONSE // ========================================== const rawResponse = await response.text(); console.log( "WORKER STATUS:", response.status ); console.log( "WORKER RESPONSE:", rawResponse ); // ========================================== // PARSE JSON // ========================================== let data; try { data = JSON.parse( rawResponse ); } catch (jsonError) { console.error( "JSON ERROR:", jsonError ); throw new Error( "Worker returned invalid JSON." ); } // ========================================== // CHECK HTTP STATUS // ========================================== if (!response.ok) { throw new Error( data.error || "Worker request failed." ); } // ========================================== // CHECK SUCCESS // ========================================== if ( data.success !== true ) { throw new Error( data.error || "Story generation failed." ); } // ========================================== // GET STORY // ========================================== const generatedStory = String( data.content || "" ).trim(); // ========================================== // CHECK EMPTY STORY // ========================================== if (!generatedStory) { throw new Error( "AI returned an empty story." ); } // ========================================== // SAVE STORY // ========================================== if ( typeof state !== "undefined" ) { state.story = generatedStory; } // ========================================== // DISPLAY STORY // ========================================== if (storyOutput) { storyOutput.textContent = generatedStory; } // ========================================== // SHOW STORY CARD // ========================================== if (storyCard) { storyCard.classList.remove( "hidden" ); storyCard.classList.add( "show" ); } // ========================================== // ACTIVATE STORY STEP // ========================================== if ( typeof activateStep === "function" ) { activateStep(2); } // ========================================== // SUCCESS STATUS // ========================================== if ( typeof showStatus === "function" ) { showStatus( "✅ Story generated successfully." ); } // ========================================== // SCROLL TO STORY // ========================================== if (storyCard) { setTimeout( function () { storyCard.scrollIntoView({ behavior: "smooth", block: "start" }); }, 100 ); } } catch (error) { // ========================================== // ERROR // ========================================== console.error( "STORY ERROR:", error ); const errorMessage = error && error.message ? error.message : "Story generation failed."; if ( typeof showStatus === "function" ) { showStatus( "❌ " + errorMessage ); } alert( "Story generation failed:\n\n" + errorMessage ); } finally { // ========================================== // RESTORE BUTTON // ========================================== if (button) { button.disabled = false; button.textContent = "✨ Generate Story"; } } }
