async function generateStory() { // ============================================ // GET USER INPUT // ============================================ const topicElement = document.getElementById("topic"); const languageElement = document.getElementById("language"); const sceneCountElement = document.getElementById("sceneCount"); const button = document.getElementById("generateStoryBtn"); const storyOutput = document.getElementById("storyOutput"); const storyCard = document.getElementById("storyCard"); // ============================================ // CHECK HTML ELEMENTS // ============================================ if (!topicElement) { alert("Topic input not found."); return; } if (!languageElement) { alert("Language selector not found."); return; } if (!sceneCountElement) { alert("Scene count selector not found."); return; } // ============================================ // READ VALUES // ============================================ const topic = topicElement.value.trim(); const language = languageElement.value; const sceneCount = Number( sceneCountElement.value ); // ============================================ // VALIDATE TOPIC // ============================================ if (!topic) { alert( "Please enter a story topic first." ); topicElement.focus(); return; } // ============================================ // SAVE STATE // ============================================ state.topic = topic; state.language = language; state.sceneCount = sceneCount; // ============================================ // BUTTON LOADING // ============================================ if (button) { button.disabled = true; button.textContent = "⏳ Generating Story..."; } // ============================================ // SHOW STATUS // ============================================ if ( typeof showStatus === "function" ) { showStatus( "Creating your story...", true ); } // ============================================ // YOUR CLOUDFLARE WORKER URL // ============================================ const WORKER_URL = "https://yt-shorts-api.nvrsarma.workers.dev/"; try { // ========================================== // SEND REQUEST TO CLOUDFLARE WORKER // ========================================== const response = await fetch( WORKER_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "story", topic: topic, language: language, sceneCount: sceneCount }) } ); // ========================================== // READ RESPONSE AS TEXT FIRST // ========================================== const rawResponse = await response.text(); console.log( "WORKER STATUS:", response.status ); console.log( "WORKER RESPONSE:", rawResponse ); // ========================================== // CONVERT RESPONSE TO JSON // ========================================== let data; try { data = JSON.parse(rawResponse); } catch (error) { throw new Error( "Worker returned invalid JSON." ); } // ========================================== // CHECK WORKER RESPONSE // ========================================== if (!response.ok) { throw new Error( data.error || "Worker request failed." ); } if (!data.success) { throw new Error( data.error || "Story generation failed." ); } // ========================================== // GET GENERATED STORY // ========================================== state.story = data.content || ""; // ========================================== // CHECK EMPTY STORY // ========================================== if ( !state.story.trim() ) { throw new Error( "AI returned an empty story." ); } // ========================================== // DISPLAY STORY // ========================================== if (storyOutput) { storyOutput.textContent = state.story; } // ========================================== // SHOW STORY CARD // ========================================== if (storyCard) { storyCard.classList.remove( "hidden" ); storyCard.classList.add( "show" ); } // ========================================== // ACTIVATE STORY STEP // ========================================== if ( typeof activateStep === "function" ) { activateStep(2); } // ========================================== // SUCCESS MESSAGE // ========================================== if ( typeof showStatus === "function" ) { showStatus( "✅ Story generated successfully." ); } // ========================================== // SCROLL TO STORY // ========================================== if (storyCard) { storyCard.scrollIntoView({ behavior: "smooth", block: "start" }); } } catch (error) { // ========================================== // ERROR HANDLING // ========================================== console.error( "STORY ERROR:", error ); if ( typeof showStatus === "function" ) { showStatus( "❌ " + ( error.message || "Story generation failed." ) ); } alert( "Story generation failed:\n\n" + ( error.message || "Unknown error" ) ); } finally { // ========================================== // RESTORE BUTTON // ========================================== if (button) { button.disabled = false; button.textContent = "✨ Generate Story"; } } }
