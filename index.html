${clean.replace(/\n/g, "
")}
`; }) .filter(Boolean); return paragraphs.join(""); } // ============================================================ // FORMAT AI CONTENT // ============================================================ function formatAIContent(text) { if (!text) return ""; return parseStoryToRichHTML(text); } // ============================================================ // APPROVE STORY // ============================================================ function approveStory() { if (!state.story) { alert( "Please generate a story first." ); CHARACTER REFERENCE SHEETS
Production-ready cinematic character prompts
${html}
`; } // ============================================================ // PARSE CHARACTERS // ============================================================ function parseCharactersToRichHTML(text) { if (!text) return ""; const lines = text.split(/\r?\n/); let html = ""; let current = null; function flushCharacter() { if (!current) return; const content = current.content .join("\n") .trim(); html += `
${escapeHTML( current.number || "★" )}
${escapeHTML( current.name || "Main Character" )}
CINEMATIC CHARACTER
${parseRichSections(content)}
`; } lines.forEach(function (line) { const trimmed = line.trim(); const match = trimmed.match( /^(?:CHARACTER|CHARACTER\s+SHEET)\s*(\d+)?\s*[:\-–]?\s*(.*)$/i ); if (match) { flushCharacter(); current = { number: match[1] || "", name: match[2] || "", content: [] }; return; } if (!current) { current = { number: "", name: "Character Reference", content: [] }; } current.content.push(line); }); flushCharacter(); if (!html.trim()) { return `
${formatAIContent(text)}
`; } return html; } // ============================================================ // RICH SECTIONS PARSER // ============================================================ function parseRichSections(text) { if (!text) return ""; const lines = text.split(/\r?\n/); let html = ""; let currentTitle = null; let currentContent = []; function flushSection() { if (!currentTitle) { if (currentContent.length) { html += `
${escapeHTML( currentContent .join("\n") ).replace( /\n/g, "
" )}
`; } return; } const content = currentContent .join("\n") .trim(); html += `
${getSectionIcon( currentTitle )} ${escapeHTML( currentTitle )}
${escapeHTML( content ).replace( /\n/g, "
" )}
`; currentTitle = null; currentContent = []; } lines.forEach(function (line) { const trimmed = line.trim(); const heading = trimmed.match( /^(?:#{1,6}\s*)?([A-Za-z][A-Za-z\s&\/-]{2,40})\s*:\s*(.*)$/ ); if (heading) { flushSection(); currentTitle = heading[1].trim(); currentContent.push( heading[2] ); return; } currentContent.push(line); }); flushSection(); return html; } // ============================================================ // GENERATE SCENES // ============================================================ async function generateScenes() { if (!state.story) { alert( "Please generate and approve the story first." ); result.sfx += cleanSectionValue( trimmed ) + " "; return; } result[currentKey] += trimmed + " "; }); return result; } // ============================================================ // CLEAN SECTION VALUE // ============================================================ function cleanSectionValue( text ) { return text .replace( /^[^:]+:\s*/, "" ) .trim(); } // ============================================================ // RENDER SCENE SECTIONS // ============================================================ function renderSceneSections( sections ) { const order = [ ["visual", "🎨", "VISUAL PROMPT"], ["characters", "👤", "CHARACTERS"], ["action", "🏃", "ACTION & MOVEMENT"], ["emotion", "❤️", "EMOTION"], ["environment", "🌎", "ENVIRONMENT"], ["camera", "🎥", "CAMERA"], ["lighting", "💡", "LIGHTING & MOOD"], ["voiceover", "🎙️", "TELUGU VOICEOVER"], ["music", "🎵", "BACKGROUND MUSIC"], ["sfx", "🔊", "SOUND EFFECTS"], ["prompt", "✨", "CINEMATIC PROMPT"] ]; ${icon} ${title}
${escapeHTML( value.trim() )}
`; }); return html; } // ============================================================ // REGENERATE STORY // ============================================================ async function regenerateStory() { const topicElement = document.getElementById( "topic" );
