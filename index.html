async function generateStory() { // ============================================ // GET ELEMENTS // ============================================ const topicElement = document.getElementById("topic"); const languageElement = document.getElementById("language"); const sceneCountElement = document.getElementById("sceneCount"); const button = document.getElementById("generateStoryBtn"); const storyOutput = document.getElementById("storyOutput"); const storyCard = document.getElementById("storyCard"); // ============================================ // CHECK ELEMENTS // ============================================ if (!topicElement) { alert("Topic input not found."); return; } if (!languageElement) { alert("Language selector not found."); return; } if (!sceneCountElement) { alert("Scene count selector not found."); return; } // ============================================ // READ INPUT // ============================================ const topic = topicElement.value.trim(); const language = languageElement.value; const sceneCount = Number(sceneCountElement.value); // ============================================ // VALIDATE // ============================================ if (!topic) { alert( "Please enter a story topic first." ); topicElement.focus(); return; } if ( !Number.isInteger(sceneCount) || sceneCount < 1 ) { alert( "Please select a valid number of scenes." ); return; } // ============================================ // SAVE STATE // ============================================ if (typeof state !== "undefined") { state.topic = topic; state.language = language; state.sceneCount = sceneCount; } // ============================================ // LOADING BUTTON // ============================================ if (button) { button.disabled = true; button.innerHTML = "⏳ Creating Story..."; } // ============================================ // STATUS // ============================================ if ( typeof showStatus === "function" ) { showStatus( "Creating your story...", true ); } // ============================================ // IMPORTANT // PLAIN URL ONLY // ============================================ const WORKER_URL = "https://yt-shorts-api.nvrsarma.workers.dev/"; try { // ========================================== // SEND TO CLOUDFLARE WORKER // ========================================== const response = await fetch( WORKER_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action: "story", topic: topic, language: language, sceneCount: sceneCount }) } ); // ========================================== // GET RAW RESPONSE // ========================================== const raw = await response.text(); console.log( "WORKER STATUS:", response.status ); console.log( "WORKER RAW:", raw ); // ========================================== // PARSE JSON // ========================================== let data; try { data = JSON.parse(raw); } catch (e) { console.error( "JSON PARSE ERROR:", e ); throw new Error( "Worker returned invalid JSON." ); } // ========================================== // HTTP ERROR // ========================================== if (!response.ok) { throw new Error( data.error || "Worker request failed." ); } // ========================================== // API ERROR // ========================================== if ( data.success !== true ) { throw new Error( data.error || "Story generation failed." ); } // ========================================== // GET CONTENT // ========================================== const story = String( data.content || "" ).trim(); // ========================================== // EMPTY CHECK // ========================================== if (!story) { throw new Error( "AI returned empty story." ); } // ========================================== // SAVE // ========================================== if ( typeof state !== "undefined" ) { state.story = story; } // ========================================== // RICH STORY DISPLAY // ========================================== if (storyOutput) { storyOutput.innerHTML = ""; const container = document.createElement( "div" ); container.className = "rich-story-result"; // ------------------------------------------ // HEADER // ------------------------------------------ const header = document.createElement( "div" ); header.className = "rich-story-header"; header.innerHTML = `
📖
Your Story
${escapeHTML(language)} • ${sceneCount} Scenes
`; container.appendChild( header ); // ------------------------------------------ // STORY BODY // ------------------------------------------ const body = document.createElement( "div" ); body.className = "rich-story-body"; body.innerHTML = formatStoryRich( story ); container.appendChild( body ); // ------------------------------------------ // FOOTER // ------------------------------------------ const footer = document.createElement( "div" ); footer.className = "rich-story-footer"; footer.innerHTML = ` 🎬 ${sceneCount} Scenes 🌐 ${escapeHTML(language)} ✨ AI Generated `; container.appendChild( footer ); storyOutput.appendChild( container ); } // ========================================== // SHOW STORY CARD // ========================================== if (storyCard) { storyCard.classList.remove( "hidden" ); storyCard.classList.add( "show" ); storyCard.style.display = "block"; } // ========================================== // STEP 2 // ========================================== if ( typeof activateStep === "function" ) { activateStep(2); } // ========================================== // SUCCESS // ========================================== if ( typeof showStatus === "function" ) { showStatus( "✅ Story generated successfully." ); } // ========================================== // SCROLL // ========================================== if (storyCard) { setTimeout( function () { storyCard.scrollIntoView({ behavior: "smooth", block: "start" }); }, 200 ); } } catch (error) { console.error( "STORY ERROR:", error ); const message = error && error.message ? error.message : "Story generation failed."; if ( typeof showStatus === "function" ) { showStatus( "❌ " + message ); } alert( "Story generation failed:\n\n" + message ); } finally { if (button) { button.disabled = false; button.innerHTML = "✨ Generate Story"; } } } // ============================================================ // RICH STORY FORMATTER // ============================================================ function formatStoryRich(text) { let html = escapeHTML(text); // Remove markdown horizontal lines html = html.replace( /^---$/gm, "" ); // H1 html = html.replace( /^# (.+)$/gm, '

' ); // H2 html = html.replace( /^## (.+)$/gm, '

$1

' ); // H3 html = html.replace( /^### (.+)$/gm, '

$1

' ); // Bold html = html.replace( /\*\*(.*?)\*\*/g, '$1' ); // Italic html = html.replace( /\*(.*?)\*/g, '$1' ); // Bullet points html = html.replace( /^\s*[-•]\s+(.+)$/gm, '
  • $1
    • ' ); // Numbered points html = html.replace( /^\s*\d+\.\s+(.+)$/gm, '
  • $1
    • ' ); // Paragraphs html = html.replace( /\n{2,}/g, "

    " ); // Single line breaks html = html.replace( /\n/g, "
    " ); html = "

    " + html + "

    "; // Clean empty paragraphs html = html.replace( /

    \s*<\/p>/g, "" ); return html; } // ============================================================ // HTML ESCAPE // ============================================================ function escapeHTML(value) { return String(value) .replace( /&/g, "&" ) .replace( //g, ">" ) .replace( /"/g, """ ) .replace( /'/g, "'" ); }
